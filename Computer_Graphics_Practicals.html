PRACTICAL II – SHOW A MOVING 2D CARTOON IN RAINY SEASON
============================================================

#include <stdlib.h>   // Contains rand() function
#include <conio.h>    // Contains kbhit() function
#include <dos.h>      // Contains delay() function
#include <graphics.h>

// Get height and width of output window
#define screenwidth getmaxx()
#define screenheight getmaxy()

// Set ground at 75% of window height
#define groundY (screenheight * 0.75)

void drawMan(int x, int legs_displacement)
{
    // Head
    circle(x, groundY - 90, 10);

    // Middle line/body
    line(x, groundY - 80, x, groundY - 30);

    // Right hand and left hand
    line(x, groundY - 70, x + 15, groundY - 60);
    line(x, groundY - 70, x - 15, groundY - 60);

    // Right leg and left leg
    line(x, groundY - 30, x + legs_displacement, groundY);
    line(x, groundY - 30, x - legs_displacement, groundY);
}

void drawUmbrella(int x)
{
    // Semi-circular top
    pieslice(x + 15, groundY - 120, 0, 180, 40);

    // Handle
    line(x + 15, groundY - 120, x + 15, groundY - 60);
}

void drawRain(int x)
{
    int i, rx, ry;

    // Draw at most 1000 rain droplets
    for (i = 0; i < 1000; i++)
    {
        rx = rand() % screenwidth;
        ry = rand() % screenheight;

        // Rain must remain above the ground and outside
        // the protected area beneath the umbrella.
        if (ry + 5 < groundY)
        {
            if (ry < groundY - 120 ||
                (ry > groundY - 120 && (rx < x - 20 || rx > x + 60)))
            {
                line(rx, ry, rx + 0.5, ry + 5);
            }
        }
    }
}

void main()
{
    int gd = DETECT, gm, x = 0, legs_displacement = 0;

    initgraph(&gd, &gm, "C://TURBOC3//bgi");

    // Set background color white and foreground color blue
    setbkcolor(WHITE);
    setcolor(BLUE);

    // Repeat until a key is pressed
    while (!kbhit())
    {
        // Draw the ground
        line(0, groundY, screenwidth, groundY);

        // Draw rain
        drawRain(x);

        // Draw man with moving legs
        legs_displacement = (legs_displacement + 2) % 20;
        drawMan(x, legs_displacement);

        // Draw umbrella
        drawUmbrella(x);

        // Wait for 75 milliseconds
        delay(75);

        // Clear screen
        cleardevice();

        // Move man and umbrella forward
        x = (x + 2) % screenwidth;
    }

    closegraph();
}

OUTPUT:
A 2D cartoon/person carrying an umbrella moves across the screen while rain falls.
Figure 4: A frame of the output – Moving 2D Cartoon in Rain.

#include <graphics.h>
#include <conio.h>
#include <iostream.h>

void dda(int xi, int yi, int xf, int yf)
{
    int dx = xf - xi;
    int dy = yf - yi;
    float m = (float)dy / dx;

    float x = xi, y = yi;
    putpixel(x, y, RED);

    if (m <= 1)
    {
        while (x < xf)
        {
            x++;
            y = y + m;
            putpixel(x, y, RED);
        }
    }
    else
    {
        while (y <= yf)
        {
            x = x + (1 / m);
            y++;
            putpixel(x, y, RED);
        }
    }
}

void main()
{
    int gd = DETECT, gm, xi, xf, yi, yf;

    initgraph(&gd, &gm, "C://TURBOC3//bgi");

    cout << "Enter value of xi, yi: ";
    cin >> xi >> yi;

    cout << "\nEnter value of xf, yf: ";
    cin >> xf >> yf;

    setbkcolor(WHITE);
    dda(xi, yi, xf, yf);

    getch();
    closegraph();
}

PRACTICAL V – IMPLEMENT BRESENHAM’S LINE GENERATING ALGORITHM

PRACTICAL V – IMPLEMENT BRESENHAM’S LINE GENERATING ALGORITHM

#include <graphics.h>
#include <conio.h>
#include <iostream.h>

void bresenham(int xi, int yi, int xf, int yf)
{
    int dx = xf - xi;
    int dy = yf - yi;
    int m = dy / dx;

    int P = 2 * dy - dx;
    int x = xi, y = yi;

    putpixel(x, y, RED);

    if (m < 1)
    {
        while (x <= xf)
        {
            if (P < 0)
            {
                x++;
                P = P + 2 * dy;
            }
            else
            {
                x++;
                y++;
                P = P + 2 * (dy - dx);
            }

            putpixel(x, y, RED);
        }
    }
    else
    {
        while (y <= yf)
        {
            if (P < 0)
            {
                y++;
                P = P + 2 * dx;
            }
            else
            {
                y++;
                x++;
                P = P + 2 * (dx - dy);
            }

            putpixel(x, y, RED);
        }
    }
}

void main()
{
    int gd = DETECT, gm, xi, xf, yi, yf;

    initgraph(&gd, &gm, "C://TURBOC3//bgi");

    cout << "Enter value of xi, yi: ";
    cin >> xi >> yi;

    cout << "\nEnter value of xf, yf: ";
    cin >> xf >> yf;

    setbkcolor(WHITE);
    bresenham(xi, yi, xf, yf);

    getch();
    closegraph();
}

#include <graphics.h>
#include <conio.h>
#include <iostream.h>

void plotPixel(int x, int y, int Xcenter, int Ycenter)
{
    // Plot the eight symmetric points of the circle
    putpixel(x + Xcenter, y + Ycenter, RED);
    putpixel(y + Xcenter, x + Ycenter, RED);
    putpixel(x + Xcenter, -y + Ycenter, RED);
    putpixel(-y + Xcenter, x + Ycenter, RED);
    putpixel(-x + Xcenter, y + Ycenter, RED);
    putpixel(y + Xcenter, -x + Ycenter, RED);
    putpixel(-x + Xcenter, -y + Ycenter, RED);
    putpixel(-y + Xcenter, -x + Ycenter, RED);
}

void drawcircle(int Xcenter, int Ycenter, int radius)
{
    int x = 0, y = radius, P = 1 - radius;

    plotPixel(x, y, Xcenter, Ycenter);

    while (x <= y)
    {
        if (P < 0)
        {
            x++;
            plotPixel(x, y, Xcenter, Ycenter);
            P = P + 2 * x + 1;
        }
        else
        {
            x++;
            y--;
            plotPixel(x, y, Xcenter, Ycenter);
            P = P + 2 * x + 1 - 2 * y;
        }
    }
}

int main()
{
    int gdriver = DETECT, gmode, P, x, y, r;

    initgraph(&gdriver, &gmode, "C://TURBOC3//bgi");

    cout << "Enter radius of circle: ";
    cin >> r;

    cout << "Enter co-ordinates of center (x and y): ";
    cin >> x >> y;

    setbkcolor(WHITE);
    drawcircle(x, y, r);

    getch();
    closegraph();
    return 0;
}

#include <graphics.h>
#include <conio.h>

void boundaryFill(int x, int y, int fillColor, int boundaryColor)
{
    int c = getpixel(x, y);

    if (c != fillColor && c != boundaryColor)
    {
        putpixel(x, y, fillColor);

        boundaryFill(x + 1, y, fillColor, boundaryColor);
        boundaryFill(x, y - 1, fillColor, boundaryColor);
        boundaryFill(x - 1, y, fillColor, boundaryColor);
        boundaryFill(x, y + 1, fillColor, boundaryColor);
    }
}

void main()
{
    int gd = DETECT, gm, x, y, fillColor, boundaryColor;

    initgraph(&gd, &gm, "C://TURBOC3//bgi");

    setbkcolor(WHITE);
    setcolor(BLUE);

    rectangle(100, 100, 158, 157);
    boundaryFill(101, 101, CYAN, BLUE);

    getch();
    closegraph();
}

#include <graphics.h>
#include <conio.h>

void floodFill(int x, int y, int fillColor, int oldColor)
{
    if (getpixel(x, y) == oldColor)
    {
        putpixel(x, y, fillColor);

        floodFill(x + 1, y, fillColor, oldColor);
        floodFill(x, y - 1, fillColor, oldColor);
        floodFill(x - 1, y, fillColor, oldColor);
        floodFill(x, y + 1, fillColor, oldColor);
    }
}

void main()
{
    int gd = DETECT, gm, x, y, fillColor, oldColor;

    initgraph(&gd, &gm, "C://TURBOC3//bgi");

    setcolor(BLUE);
    rectangle(100, 100, 150, 150);

    floodFill(101, 101, CYAN, BLACK);

    getch();
    closegraph();
}

PRACTICAL IX – LINE CLIPPING USING COHEN-SUTHERLAND ALGORITHM

PRACTICAL IX – LINE CLIPPING USING COHEN-SUTHERLAND ALGORITHM
============================================================

#include <iostream.h>
#include <graphics.h>
#include <conio.h>

// Define region codes
const int INSIDE = 0;  // 0000
const int LEFT = 1;    // 0001
const int RIGHT = 2;   // 0010
const int BOTTOM = 4;  // 0100
const int TOP = 8;     // 1000

// Clipping rectangle boundaries
const int x_max = 400;
const int y_max = 200;
const int x_min = 100;
const int y_min = 100;

int computeCode(double x, double y)
{
    // Initialize point as inside
    int code = INSIDE;

    if (x < x_min)
        code = code | LEFT;
    else if (x > x_max)
        code = code | RIGHT;

    if (y < y_min)
        code = code | BOTTOM;
    else if (y > y_max)
        code = code | TOP;

    return code;
}

void cohenSutherlandClip(int x1, int y1, int x2, int y2)
{
    int code1 = computeCode(x1, y1);
    int code2 = computeCode(x2, y2);
    int accept = 0;

    while (1)
    {
        if ((code1 == 0) && (code2 == 0))
        {
            // Both endpoints are inside
            accept = 1;
            break;
        }
        else if (code1 & code2)
        {
            // Both endpoints are outside in the same region
            break;
        }
        else
        {
            // Some part of the line lies inside the rectangle
            int code_out;
            int x, y;

            if (code1 != 0)
                code_out = code1;
            else
                code_out = code2;

            // Find intersection point
            if (code_out & TOP)
            {
                x = x1 + (x2 - x1) * (y_max - y1) / (y2 - y1);
                y = y_max;
            }
            else if (code_out & BOTTOM)
            {
                x = x1 + (x2 - x1) * (y_min - y1) / (y2 - y1);
                y = y_min;
            }
            else if (code_out & RIGHT)
            {
                y = y1 + (y2 - y1) * (x_max - x1) / (x2 - x1);
                x = x_max;
            }
            else if (code_out & LEFT)
            {
                y = y1 + (y2 - y1) * (x_min - x1) / (x2 - x1);
                x = x_min;
            }

            // Replace the outside point with intersection point
            if (code_out == code1)
            {
                setcolor(RED);
                line(x1, y1, x, y);
                x1 = x;
                y1 = y;
                code1 = computeCode(x1, y1);
            }
            else
            {
                setcolor(RED);
                line(x2, y2, x, y);
                x2 = x;
                y2 = y;
                code2 = computeCode(x2, y2);
            }
        }
    }

    if (accept)
    {
        setcolor(GREEN);
        line(x1, y1, x2, y2);
    }
    else
    {
        setcolor(RED);
        line(x1, y1, x2, y2);
    }
}

void main()
{
    int gd = DETECT, gm;

    initgraph(&gd, &gm, "C://TURBOC3//bgi");

    setbkcolor(WHITE);
    setcolor(BLUE);

    rectangle(x_min, y_min, x_max, y_max);

    cohenSutherlandClip(75, 75, 150, 150);
    cohenSutherlandClip(250, 50, 250, 250);
    cohenSutherlandClip(425, 110, 350, 190);

    getch();
    closegraph();
}

OUTPUT:
A clipping rectangle is displayed. Green lines indicate accepted lines, and red lines indicate the clipped/rejected portions.

#include <graphics.h>
#include <conio.h>

void drawLine(int P[][2], int color)
{
    setcolor(color);
    line(P[0][0], P[0][1], P[1][0], P[1][1]);
}

void translateLine(int P[][2], int T[])
{
    P[0][0] = P[0][0] + T[0];
    P[0][1] = P[0][1] + T[1];

    P[1][0] = P[1][0] + T[0];
    P[1][1] = P[1][1] + T[1];
}

void main()
{
    int P[2][2] = {{100, 100}, {200, 200}};
    int T[2] = {100, 10};
    int gd = DETECT, gm;

    initgraph(&gd, &gm, "C://TURBOC3//bgi");

    setbkcolor(WHITE);

    drawLine(P, BLUE);
    translateLine(P, T);
    drawLine(P, RED);

    getch();
    closegraph();
}

OUTPUT:
The original line is blue and the translated line is red.
